How to use it on your site

// Search eBay sold listings
fetch('https://ebayapidata.onrender.com/scrape/search?query=mahomes+prizm', {
  headers: { 'X-API-Key': 'YOUR_API_KEY_HERE' }
})
.then(r => r.json())
.then(listings => {
  listings.forEach(item => {
    console.log(item.title, item.sale_price, item.sale_date);
  });
});
